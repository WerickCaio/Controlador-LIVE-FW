void melquisedeque() // Vai ficar alterando o valor de SS, para que o cÃ³digo rode de forma decente
{
    uint8_t state = PORTB & DS;
    if (state)
    {
        PORTB &= ~DS;
    }
    else
    {
        PORTB |= DS;
    }
}

void toggleOE() // Vai ficar alterando o valor de OE, para que o cÃ³digo rode de forma decente
{
    uint8_t state = (PORTD & OE);
    // Serial.println(state>>5, BIN);
    Serial.println(state);
    if (state >> 5)
    {
        PORTD &= ~OE;
    }
    else
    {
        PORTD |= OE;
    }
}

void desativaTudo()
{
    PORTD |= B;
    PORTD |= A;
}

void readPlacarEEPROM()
{
    uint8_t lowBytePart;
    uint8_t highBytePart;

    for (int i = 0; i < 6; i++)
    {
        lowBytePart = EEPROM.read(i * 2);
        highBytePart = EEPROM.read((i * 2) + 1);
        teamsScore[i] = lowBytePart + highBytePart * 256;
    }
}

void savePlacarEEPROM()
{
    uint8_t lowBytePart;
    uint8_t highBytePart;

    for (int i = 0; i < 6; i++)
    {
        lowBytePart = teamsScore[i] & 0b11111111;
        highBytePart = (teamsScore[i] >> 8) & 0b11111111;

        EEPROM.write(i * 2, lowBytePart);
        EEPROM.write((i * 2) + 1, highBytePart);
    }
}

void clearPlacarEEPROM()
{
    for (int i = 0; i < 12; i++)
    {
        EEPROM.write(i, 0);
    }

    for (int i = 0; i < 6; i++)
    {
        teamsScore[i] = 0;
    }
    savePlacarEEPROM(); // <- garante que o zero foi salvo antes do reset
}
