int Comm_GetCommand() {
    // 1. USB Serial (cabo)
    if (Serial.available() > 0) {
        String entrada = Serial.readStringUntil('\n');
        entrada.trim();
        if (entrada.length() == 0) return -1;

        int cmd = entrada.toInt();
        if (cmd == 0 && entrada != "0") {
            DEBUG_PRINTLN("[Serial] Comando desconhecido ignorado.");
            return -1;
        }
        DEBUG_PRINTF("[Serial] Comando recebido: %d\n", cmd);
        return cmd;
    }

    // 2. Bluetooth ClÃ¡ssico â€” descomente se usar Serial1:
    /*

    DEBUG_PRINTLN("[Scoreboard] Placar zerado e salvo.");
}

void Scoreboard_DrawBoxes() {
    int tpr   = teamsPerRow();
    int nRows = numRows();

    for (int row = 0; row < nRows; row++) {
        int yTop = rowBoxTop(row);
        int yBot = rowBoxBottom(row);

        // Quantos times nessa linha? (Ãºltima linha pode ter menos)
        int teamsThisRow = min(tpr, NUM_TEAMS - row * tpr);

        for (int col = 0; col < teamsThisRow; col++) {
            int xL = boxX_Start(col);
            int xR = boxX_End(col);

            // Bordas horizontais (topo e base)
            for (int x = xL; x <= xR; x++) {
                Display_PutPixel(x, yTop, cyan);
                Display_PutPixel(x, yBot, cyan);
            }
            // Bordas verticais (esquerda e direita)
            for (int y = yTop + 1; y < yBot; y++) {
                Display_PutPixel(xL, y, cyan);
                Display_PutPixel(xR, y, cyan);
            }
        }
    }
}

}

{

    pinMode(PIN_LAT, OUTPUT);
    pinMode(PIN_OE, OUTPUT);
    pinMode(PIN_CLK, OUTPUT);

    EEPROM.begin(512);
    SPI.begin(PIN_CLK, -1, PIN_DATA, -1);
}

int HAL_PingHardware()
{
    pinMode(PIN_DATA, OUTPUT);
    pinMode(PIN_CLK, OUTPUT);
    pinMode(PIN_OE, OUTPUT);
    pinMode(PIN_RETORNO, INPUT_PULLDOWN);

    digitalWrite(PIN_OE, HIGH);
    digitalWrite(PIN_CLK, LOW);

    digitalWrite(PIN_DATA, LOW);
    for (int i = 0; i < 20000; i++)
    {
        digitalWrite(PIN_CLK, HIGH);
        digitalWrite(PIN_CLK, LOW);
    }

    digitalWrite(PIN_DATA, HIGH);
    digitalWrite(PIN_CLK, HIGH);
    digitalWrite(PIN_CLK, LOW);
    digitalWrite(PIN_DATA, LOW);

    int contador_de_bits = 1;
    while (digitalRead(PIN_RETORNO) == LOW && contador_de_bits < 20000)
    {
        digitalWrite(PIN_CLK, HIGH);
        digitalWrite(PIN_CLK, LOW);
        contador_de_bits++;
    }

    SPI.end();
    SPI.begin(PIN_CLK, -1, PIN_DATA, -1);
    return contador_de_bits;
}

#include <Arduino.h>

// Novo mapeamento dos pinos
const int PIN_R1 = 11;
const int PIN_G1 = 12;
const int PIN_B1 = 10;
const int PIN_R2 = 7;
const int PIN_G2 = 15;
const int PIN_B2 = 6;

const int PIN_A = 18;
const int PIN_B = 8;

const int PIN_CLK = 9;
const int PIN_LAT = 46;
const int PIN_OE = 3;

// Array para facilitar a inicializaÃ§Ã£o
const int hubPins[] = {PIN_R1, PIN_G1, PIN_B1, PIN_R2, PIN_G2, PIN_B2,
                       PIN_A, PIN_B, PIN_CLK, PIN_LAT, PIN_OE};
const int numPins = sizeof(hubPins) / sizeof(hubPins[0]);

void setup()
{
    Serial.begin(115200);
    delay(1000);
    Serial.println("\n--- Iniciando Sistema de Teste HUB75 ---");

    // Inicializa todos os pinos como saÃ­da com nÃ­vel lÃ³gico 0
    for (int i = 0; i < numPins; i++)
    {
        pinMode(hubPins[i], OUTPUT);
        digitalWrite(hubPins[i], LOW);
    }

    Serial.println("Todos os pinos configurados como OUTPUT em nivel LOW (0V).");
    Serial.println("Envie comandos no formato: PINO ESTADO (Exemplo: R1 1, OE 0, CLK 1)");
    Serial.println("----------------------------------------");

    delay(5000);
    Serial.println("Indo para o LOOP agora");
}

void processaComando(String comando)
{
    // Limpa espaÃ§os extras e caracteres de nova linha
    comando.trim();
    comando.toUpperCase();

    // Se estiver vazio (ex: usuÃ¡rio apenas apertou Enter sem querer), ignora
    if (comando.length() == 0)
        return;

    // Feedback imediato indicando que o ESP32 recebeu a string
    Serial.print("Comando recebido: [");
    Serial.print(comando);
    Serial.print("] -> ");

    int espacoIndex = comando.indexOf(' ');

    // Verifica se o formato tem um espaÃ§o separando pino e estado
    if (espacoIndex == -1)
    {
        Serial.println("Comando incorreto! Falta o espaco entre o pino e o estado.");
        return;
    }

    String pinoStr = comando.substring(0, espacoIndex);
    int estado = comando.substring(espacoIndex + 1).toInt();

    int pinoFisico = -1;

    // Mapeamento das strings para os pinos fÃ­sicos
    if (pinoStr == "R1")
        pinoFisico = PIN_R1;
    else if (pinoStr == "G1")
        pinoFisico = PIN_G1;
    else if (pinoStr == "B1")
        pinoFisico = PIN_B1;
    else if (pinoStr == "R2")
        pinoFisico = PIN_R2;
    else if (pinoStr == "G2")
        pinoFisico = PIN_G2;
    else if (pinoStr == "B2")
        pinoFisico = PIN_B2;
    else if (pinoStr == "A")
        pinoFisico = PIN_A;
    else if (pinoStr == "B")
        pinoFisico = PIN_B;
    else if (pinoStr == "CLK")
        pinoFisico = PIN_CLK;
    else if (pinoStr == "LAT")
        pinoFisico = PIN_LAT;
    else if (pinoStr == "OE")
        pinoFisico = PIN_OE;

    // Aplica o comando se o pino foi reconhecido
    if (pinoFisico != -1)
    {
        digitalWrite(pinoFisico, estado > 0 ? HIGH : LOW);
        Serial.print("Sucesso! Pino ");
        Serial.print(pinoStr);
        Serial.print(" alterado para ");
        Serial.println(estado > 0 ? "HIGH" : "LOW");
    }
    else
    {
        Serial.println("Comando incorreto! O nome do pino nao existe na lista.");
    }
}

void loop()
{
    if (Serial.available() > 0)
    {
        String comando = Serial.readStringUntil('\n');
        processaComando(comando);
    }
}
